"use client";

import { useState } from "react";

type Message = {
  role: "user" | "assistant";
  content: string;
};

export default function HomePage() {
  const [messages, setMessages] = useState<Message[]>([]);
  const [input, setInput] = useState("");
  const [isLoading, setIsLoading] = useState(false);

  async function handleSubmit(e: React.FormEvent) {
    e.preventDefault();
    if (!input.trim() || isLoading) return;

    const newMessage: Message = { role: "user", content: input.trim() };
    const newMessages = [...messages, newMessage];

    setMessages(newMessages);
    setInput("");
    setIsLoading(true);

    try {
      const res = await fetch("/api/chat", {
        method: "POST",
        body: JSON.stringify({ messages: newMessages }),
        headers: {
          "Content-Type": "application/json",
        },
      });

      // Streaming "manual" simple (sin librería especial en el cliente)
      const reader = res.body?.getReader();
      const decoder = new TextDecoder();
      let assistantText = "";

      if (!reader) throw new Error("No hay body en la respuesta");

      while (true) {
        const { done, value } = await reader.read();
        if (done) break;
        const chunk = decoder.decode(value);
        assistantText += chunk;

        setMessages((prev) => {
          // Reemplazamos el último mensaje del asistente mientras se streamea
          const last = prev[prev.length - 1];
          if (last && last.role === "assistant") {
            return [
              ...prev.slice(0, prev.length - 1),
              { role: "assistant", content: assistantText },
            ];
          } else {
            return [...prev, { role: "assistant", content: assistantText }];
          }
        });
      }
    } catch (err) {
      console.error(err);
      setMessages((prev) => [
        ...prev,
        {
          role: "assistant",
          content:
            "Hubo un error al hablar con el asistente. Probá de nuevo más tarde.",
        },
      ]);
    } finally {
      setIsLoading(false);
    }
  }

  return (
    <main className="min-h-screen flex flex-col items-center justify-between p-4 md:p-8">
      <div className="w-full max-w-2xl flex flex-col gap-4">
        <h1 className="text-2xl font-bold">📚 AI Book Advisor</h1>

        <div className="border rounded-lg p-4 h-[60vh] overflow-y-auto bg-white/5">
          {messages.length === 0 && (
            <p className="text-sm text-gray-500">
              Empezá con algo como:{" "}
              <strong>"Recomendame libros de ciencia ficción"</strong>
            </p>
          )}
          {messages.map((m, i) => (
            <div
              key={i}
              className={`mb-2 ${
                m.role === "user" ? "text-right" : "text-left"
              }`}
            >
              <span
                className={
                  m.role === "user" ? "font-semibold text-blue-400" : "font-semibold text-green-400"
                }
              >
                {m.role === "user" ? "Vos" : "Asistente"}
              </span>
              <p className="whitespace-pre-wrap">{m.content}</p>
            </div>
          ))}
          {isLoading && (
            <p className="text-sm text-gray-400 mt-2">Pensando / buscando libros…</p>
          )}
        </div>

        <form onSubmit={handleSubmit} className="flex gap-2">
          <input
            className="flex-1 border rounded-lg px-3 py-2"
            placeholder="Escribí tu pregunta sobre libros..."
            value={input}
            onChange={(e) => setInput(e.target.value)}
          />
          <button
            type="submit"
            disabled={isLoading}
            className="px-4 py-2 rounded-lg border font-semibold disabled:opacity-50"
          >
            {isLoading ? "Enviando..." : "Enviar"}
          </button>
        </form>
      </div>
    </main>
  );
}
